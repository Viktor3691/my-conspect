Примеры использования if-else в JavaScript

1. Проверка числа на положительность

script.js 
let num = 10;

if (num > 0) {
      console.log('Число положительное');
} else if (num < 0) {
      console.log('Число отрицательное');
 } else {
      console.log('Число равно нулю');
}
Число положительное

2. Проверка возраста (совершеннолетие)

script.js 
let age = 17;

if (age >= 18) {
    console.log('Доступ разрешён');
} else {
    console.log('Доступ запрещён');
}
Доступ запрещён

3. Определение чётности числа

script.js 
let num = 7;

if (num % 2 === 0) {
console.log('Число чётное');
} else {
console.log('Число нечётное');
}
Число нечётное

4. Оценка по баллам (несколько условий)

script.js 
let score = 85;

if (score >= 90) {
console.log('Отлично');

} else if (score >= 75) {
console.log('Хорошо');

} else if (score >= 60) {
console.log('Удовлетворительно');

} else {
console.log('Неудовлетворительно');
}
Хорошо

5. Проверка строки на пустоту

script.js 
let name = 'Виктор';

if (name === '') {
console.log('Имя не введено');

} else {
console.log('Привет, ' + name);
}
Привет, Виктор

6. Проверка диапазона (декада месяца)

script.js 
let day = 25;

if (day >= 1 && day <= 10) {
    console.log('Первая декада');

} else if (day> 10 && day <= 20) {
    console.log('Вторая декада');

} else if (day> 20 && day <= 31) {
    console.log('Третья декада');

} else {
    console.log('Некорректное значение дня');
}
Третья декада

7. Вложенные if (проверка числа и его суммы цифр)

script.js 
let num = 55;

if (num < 10 || num> 99) {
    console.log('Число не попадает в диапазон от 10 до 99');

} else {
    const str = String(num);
    const sum = +str[0] + +str[1];

    if (sum <= 9) {
      console.log('Сумма цифр однозначна');

    } else {
      console.log('Сумма цифр двузначна');
 }
}
Сумма цифр двузначна

8. Проверка типа данных (строгое равенство)

script.js 
let value = '1';

if (value === 1) {
    console.log('Это число 1');

} else {
    console.log('Это не число 1 (или не строгое равенство)');
}
Это не число 1 (или не строгое равенство)

9. Проверка на существование (undefined / null)

script.js 
let user = null;

if (user === null) {
    console.log('Пользователь не найден');

} else {
    console.log('Пользователь: ' + user);
}
Пользователь не найден
script.js 
let user91;

if (user91 === undefined) {
    console.log('Пользователь не найден');

} else {
    console.log('Пользователь: ' + user91);
}
Пользователь не найден

10. Реальный сценарий — проверка формы (валидация)

script.js 
let email = 'test@mail.ru';
let password = '123321';

if (email === '' || password === '') {
    console.log('Заполните все поля');

} else if (password.length < 6) {
    console.log('Пароль слишком короткий');

} else {
    console.log('Форма отправлена');
}

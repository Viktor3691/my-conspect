Поиск ошибок в коде с условиями JavaScript

В следующих задачах некоторый программист написал код и, возможно, допустил в нем ошибки. Вы должны проверить, делает ли код то, что описано. Если код работает некорректно, вы должны исправить ошибки.

Задача 1

Код должен проверить сумму чисел:

script.js 
let num11 = 1;
let num12 = 2;

if (num11 + num12 === 3) {
console.log('+++');

} else {
console.log('---');
}
+++

Задача 14

Код должен проверить равна ли сумма первых трех цифр числа сумме вторых трех чисел:

script.js 
let num141 = '123033'; // берем в кавычки, чтобы обращаться к цифрам

let sum141 = num141[0] + num141[1] + num141[2];
let sum142 = num141[3] + num141[4] + num141[5];

if (sum141 == sum142) {
console.log('суммы равны');

} else {
console.log('суммы не равны');
}

Решение

script.js 
let num141 = '123033'; // берем в кавычки, чтобы обращаться к цифрам

let sum141 = Number(num141[0]) + Number(num141[1]) + Number(num141[2]);
let sum142 = Number(num141[3]) + Number(num141[4]) + Number(num141[5]);

if (sum141 == sum142) {
console.log('суммы равны');

} else {
console.log('суммы не равны');
}

Практика на циклы с массивами и объектами JavaScript

Задача 1

Дан следующий объект с работниками и их зарплатами:

script.js 
let obj = {
employee1: 100,
employee2: 200,
employee3: 300,
employee4: 400,
employee5: 500,
employee6: 600,
employee7: 700,
};

Увеличьте зарплату каждого работника на 10%.

script.js 

let obj1 = {
employee1: 100,
employee2: 200,
employee3: 300,
employee4: 400,
employee5: 500,
employee6: 600,
employee7: 700,
};

for (let key in obj1) {
    obj1[key] = obj1[key] + ((obj1[key] * 10) / 100);
}

console.log(obj1);
{
employee1: 110,
employee2: 220,
employee3: 330,
employee4: 440,
employee5: 550,
employee6: 660,
employee7: 770
}

Задача 2

Модифицируйте предыдущую задачу так, чтобы зарплата увеличивалась только тем работникам, у которых она меньше или равна 400.

script.js 

let obj2 = {
employee1: 100,
employee2: 200,
employee3: 300,
employee4: 400,
employee5: 500,
employee6: 600,
employee7: 700,
};

for (let key in obj2) {
    if (obj2[key] <= 400)
        obj2[key] = obj2[key] + ((obj2[key] * 10) / 100);
}

console.log(obj2);
{employee1: 110, employee2: 220, employee3: 330, employee4: 440, employee5: 500, …}

Задача 3

Даны следующие массивы:

let arr1 = [1, 2, 3, 4, 5];

let arr2 = [6, 7, 8, 9, 10];

С помощью этих массивов создайте новый объект, сделав его ключами элементы первого массива, а значениями - элементы второго.

script.js 
let arrKeys = [1, 2, 3, 4, 5];
let arrValues = [6, 7, 8, 9, 10];
let obj3 = {};

for (let i = 0; i <= 4; i++) {
let key = arrKeys[i];
let value = arrValues[i];
obj3[key] = value;
}
console.log(obj3);
{1: 6, 2: 7, 3: 8, 4: 9, 5: 10}

Задача 4

Дан следующий объект:

let obj = {1: 6, 2: 7, 3: 8, 4: 9, 5: 10};

Найдите сумму ключей этого объекта и поделите ее на сумму значений.

script.js 
let obj4 = { 1: 6, 2: 7, 3: 8, 4: 9, 5: 10 };

let sumKey = 0;
let sumValue = 0;
let drob;

for (let key in obj4) {
sumKey = sumKey + Number(key);
console.log(sumKey);
sumValue = sumValue + Number(obj4[key]);
console.log(sumValue);
drob = sumKey / sumValue;
}

console.log(drob);
0.375

Задача 6

Дан следующий объект:

script.js 
let obj = {

1: 125,
2: 225,
3: 128,
4: 356,
5: 145,
6: 281,
7: 452,
};

Запишите в новый массив элементы, значение которых начинается с цифры 1 или цифры 2. То есть у вас в результате получится вот такой массив:

script.js 
let arr = [
125,
225,
128,
145,
281,
];

Решение

script.js 
let obj6 = {
    1: 125,
    2: 225,
    3: 128,
    4: 356,
    5: 145,
    6: 281,
    7: 452,
};

let arr6 = [];

for (let key in obj6) {
    let value = obj6[key];
    let numOne = String(value)[0];

    if (numOne === '1' || numOne === '2') {
         arr6.push(value);
    }
}

document.write('arr6 = [', arr6, '];',);

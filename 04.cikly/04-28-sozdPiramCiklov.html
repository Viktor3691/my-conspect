Вывод пирамидок на JavaScript

Давайте теперь выведем на экран следующую пирамидку:

x
xx
xxx
xxxx
xxxxx
Приступим к решению задачи.

Для начала давайте вспомним код из предыдущих уроков, выводящий строку 'xxxxxxxxxx':

script.js 
let str = '';

for (let i = 0; i < 10; i++) {
str +='x' ;
}
document.write(str);
// выведет 'xxxxxxxxxx'

Для решения поставленной задачи на пирамидку просто модифицируем приведенный код: при каждом проходе цикла будем выводить на экран содержимое переменной str и тег <br>:

script.js 
let str = '';

for (let i = 0; i < 5; i++) {
str +='x' ;
document.write(str + '<br>' );
}

Так как в цикле str постепенно растет, то сначала выведется x и <br>, потом xx и <br> и так далее.

Задача 1

Выведите на экран следующую пирамидку:

xx
xxxx
xxxxxx
xxxxxxxx
xxxxxxxxxx

Решение

script.js 
let str1 = '';

for (let i = 0; i < 5; i++) {
    str1 = str1 + 'x' + 'x';

    document.write(str1,'<br>');
}

Пирамидка с цифрами

Давайте теперь выведем на экран следующую пирамидку:

111
222
333
444
555
666
777
888
999

Похожую задачу мы уже решали, когда выводили строку '111222333444555666777888999'. Напомню этот код:

script.js 
for (let i = 1; i <= 9; i++) {
for (let j=1; j <=3; j++) {
document.write(i);
// выводит '111' , потом '222', потом '333' и так далее
}
}

По сути нам нужна такая же строка, но с <br> в нужных местах. Добавим их:

script.js 
for (let i = 1; i <= 9; i++) {
for (let j=1; j <=3; j++) {
document.write(i);
}
document.write('<br>');
// выводим <br>, чтобы было в столбик
}

Выведем теперь следующую пирамидку:

1
22
333
4444
55555
666666
7777777
88888888
999999999

Задача аналогична предыдущей, только внутренний цикл должен крутиться не 3 раза, а в соответствии с числом: если у нас единица - то повторяем цикл 1 раз, если двойка - два раза, если тройка - три раза.

Как этого достичь: нужно в условии внутреннего цикла написать следующее: j <= i.
В этом случае если i равен 1, то это все равно, что в условии написать j <=1, а если i равен 2 - то j <=2 и так далее:

script.js 
for (let i=1; i <=9; i++) {
for (let j=1;j <=i; j++) {
    document.write(i);
    }
document.write('<br>');
}

Я думаю, вы уже обратили внимание, что в этой пирамидке мы не используем накопление в строку, которое мы разбирали ранее. Можно и с накоплением, но это получится чуть сложнее:

script.js 
for (let i = 1; i <= 9; i++) {
let str='' ;
// каждый раз зачищаем строку

for (let j=1; j <=i; j++) {
    str +=i;
}
document.write(str + '<br>' );
}

Задача 2

Выведите на экран следующую пирамидку: 
1
333
55555
7777777
999999999

Решение

script.js 
for (let i = 1; i <= 9; i++) {
    let str = '';

    if (i % 2 !== 0) {
        for (let j = 1; j <= i; j++) {
            str = str + i;
        }
        document.write(str, '<br>');
    }
}

Перевернутые пирамидки

Научимся теперь делать перевернутые пирамидки. Сделаем, к примеру, вот такую: 

xxxxx
xxxx
xxx
xx
x
Для решения задачи необходимо запустить два вложенных цикла, вот так:

script.js 
for (let i=5; i>= 1; i--) {
    let str = '';
    for (let j = 0; j < i; j++) {
        str +='x' ;
    }
    document.write(str + '<br>' );
}

Задача 3

Выведите на экран следующую пирамидку: 

xxxxxxxxxx
xxxxxxxx
xxxxxx
xxxx
xx

Решение

script.js 
for (let i = 10; i >= 2; i--) {
    let str3 = '';

    if (i % 2 === 0) {
        for (let j = 0; j < i; j++) {
             str3 += 'x';
        }
        document.write(str3,'<br>');
    }
}

Задача 4

Выведите на экран следующую пирамидку: 

999999999
88888888
7777777
666666
55555
4444
333
22
1

Решение

script.js 
for (let i = 9; i >= 1; i--) {
    let str = '';

    for (let j = 1; j <= i; j++) {
        str += i
    }

    document.write(str, '<br>');
}

Поиск ошибок в коде с циклами JavaScript

В следующих задачах некоторый программист написал код и, возможно, допустил в нем ошибки. Вы должны проверить, делает ли код то, что описано. Если код работает некорректно, вы должны исправить ошибки.

Задача 1

script.js 
for (let i = 0; i > 10; i++) {
console.log(i);
}

Решение

script.js 
for (let i = 0; i < 10; i++) {
// i > 10;
document.write(i, ', ');
}

Задача 2

Код должен вывести числа от 10 до 0:

script.js 
for (let i = 10; i > 0; i++) {
console.log(i);
}

Решение

script.js 
for (let i = 10; i >= 0; i--) {
//i++
    document.write(i, ', ');
}

Задача 13

Код должен возвести в квадрат каждый элемент массива:

script.js 
let arr=[1, 2, 3, 4, 5];

for (let elem of arr) {
elem=elem ** 2;
}

console.log(arr);

Решение

В коде массив не меняется, потому что в цикле for...of переменная elem — это просто копия значения элемента, а не ссылка на ячейку массива. Когда elem = elem ** 2, меняешь только эту локальную переменную, а сам массив остаётся прежним.

Как исправить через цикл с индексом (чтобы менять исходный массив)

script.js 
let arr13 = [1, 2, 3, 4, 5];

for (let i = 0; i < arr13.length; i++) {
    arr13[i] = arr13[i] ** 2;
}

document.write('arr13 = [', arr13, '];');

Задача 17

Код должен проверить, есть ли в массиве число 3 или нет:

script.js 
let arr=[1, 2, 3, 4, 5];
let res='';

for (let elem of arr) {
if (elem===3) {
res='+++' ;

} else {
res='---' ;
}
}
console.log(res);

Решение

Почему код не работает

  • elem = 1 → res = '---'
  • elem = 2 → res = '---'
  • elem = 3 → res = '+++'
  • elem = 4 → res = '---' ← тут значение снова меняется
  • elem = 5 → res = '---' ← финальное значение
script.js 
let arr17 = [1, 2, 3, 4, 5];
let res17 = '';

for (let elem17 of arr17) {
    if (elem17 === 3) {
         res17 = '+++';
         break; // как только нашли 3, выходим из цикла
    } else {
         res17 = '---';
    }
}
document.write(res17);
